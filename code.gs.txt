function doGet(e) {
  var type = (e && e.parameter && e.parameter.type) ? e.parameter.type : "nap";
  
  // ---------------- SERVER-SIDE CACHING (1. Cache Check) ----------------
  var cache = CacheService.getScriptCache();
  var cachedData = cache.get("cache_" + type);
  if (cachedData) {
    return ContentService.createTextOutput(cachedData)
      .setMimeType(ContentService.MimeType.JSON);
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var resultData = null;

  if (type === "lcp") {
    // ---------------- LCP DATA ----------------
    var lcpSheet = ss.getSheetByName("NLZ LCP Report");
    if (!lcpSheet) {
      resultData = { lcpAging: [], lcpImpact: [] };
    } else {
      // 1. LCP Aging (G24:L30)
      var agingRaw = lcpSheet.getRange("G24:L30").getValues();
      var lcpAging = [];
      for (var i = 0; i < agingRaw.length; i++) {
        var rowA = agingRaw[i];
        var area = rowA[0] ? String(rowA[0]).trim() : "";
        if (area !== "" && area.toUpperCase() !== "TOTAL") {
          lcpAging.push({
            "A": area,      // AREA
            "P": rowA[1],   // PROVINCE
            "H": rowA[2],   // <24HOURS
            "D1": rowA[3],  // 1-3 DAYS
            "D3": rowA[4],  // >3DAYS
            "T": rowA[5]    // TOTAL
          });
        }
      }

      // 2. LCP Impact (G2:K18)
      var impactRaw = lcpSheet.getRange("G2:K18").getValues();
      var lcpImpact = [];
      for (var j = 1; j < impactRaw.length; j++) {
        var rowI = impactRaw[j];
        var impArea = rowI[0] ? String(rowI[0]).trim() : "";
        if (impArea !== "" && impArea.toUpperCase() !== "TOTAL") {
          lcpImpact.push({
            "A": impArea,   // AREA
            "P": rowI[1],   // PROVINCE
            "TT": rowI[2],  // TT COUNT
            "LCP": rowI[3], // LCP COUNT
            "C": rowI[4]    // CLIENTS
          });
        }
      }
      resultData = { "lcpAging": lcpAging, "lcpImpact": lcpImpact };
    }

  } else if (type === "olt") {
    // ---------------- OLT DATA ----------------
    var oltSheet = ss.getSheetByName("NLZ OLT Report");
    if (!oltSheet) {
      resultData = [];
    } else {
      var lastRowOlt = oltSheet.getLastRow();
      if (lastRowOlt < 3) {
        resultData = [];
      } else {
        var oltData = oltSheet.getRange(3, 2, lastRowOlt - 2, 12).getValues(); 
        var oltList = [];

        var agingMap = {};
        var agingMapByName = {};
        var ticketSheet = ss.getSheetByName("OLT DOWN Tickets");
        if (ticketSheet) {
          var lastRowTix = ticketSheet.getLastRow();
          if (lastRowTix >= 2) {
            var tixData = ticketSheet.getRange(2, 1, lastRowTix - 1, 28).getValues();
            for (var t = 0; t < tixData.length; t++) {
              var ticketNoRaw = tixData[t][5];   // Column F = TICKET
              var agingRaw = tixData[t][23];     // Column X = AGING DURATION
              var oltNameRaw = tixData[t][27];   // Column AB = OLT NAME
              var agingVal = agingRaw ? String(agingRaw).trim() : "-";

              if (ticketNoRaw && String(ticketNoRaw).trim() !== "") {
                agingMap[String(ticketNoRaw).trim().toUpperCase()] = agingVal;
              }
              if (oltNameRaw && String(oltNameRaw).trim() !== "") {
                agingMapByName[String(oltNameRaw).trim().toUpperCase()] = agingVal;
              }
            }
          }
        }

        for (var k = 0; k < oltData.length; k++) {
          var rowO = oltData[k];
          var oltName = rowO[2];

          if (oltName && String(oltName).trim() !== "" && String(oltName).toUpperCase() !== "TOTAL") {
            var status = "UP";
            var ticketNo = "N/A";

            if (rowO[4] === true) {
              status = "DOWN";
              ticketNo = rowO[8] || "N/A";
            } else if (rowO[5] === true) {
              status = "OLT UPLINK LOW POWER";
              ticketNo = rowO[9] || "N/A"; // Column K
            } else if (rowO[6] === true) {
              status = "OLT UPLINK DOWN";
              ticketNo = rowO[10] || "N/A"; // Column L
            } else if (rowO[7] === true) {
              status = "OLT SERVICE DEGRADATION";
              ticketNo = rowO[11] || "N/A"; // Column M
            }

            var aging = "-";
            if (status !== "UP") {
              // 1. Subukan munang tugmain gamit ang Ticket No
              if (ticketNo && ticketNo !== "N/A") {
                var agingKey = String(ticketNo).trim().toUpperCase();
                if (agingMap.hasOwnProperty(agingKey)) {
                  aging = agingMap[agingKey];
                }
              }
              // 2. Kung walang match, i-fallback sa OLT Name (Column AB)
              if (aging === "-") {
                var nameKey = String(oltName).trim().toUpperCase();
                if (agingMapByName.hasOwnProperty(nameKey)) {
                  aging = agingMapByName[nameKey];
                }
              }
            }

            oltList.push({
              "P": rowO[0],
              "M": rowO[1],
              "N": oltName,
              "S": status,
              "T": ticketNo,
              "AG": aging
            });
          }
        }
        resultData = oltList;
      }
    }

  } else {
    // ---------------- NAP DATA (DEFAULT) ----------------
    var napSheet = ss.getSheetByName("NLZ NAP Report");
    if (!napSheet) {
      resultData = [];
    } else {
      var napRaw = napSheet.getRange("H2:M19").getValues(); 
      var napList = [];

      for (var r = 1; r < napRaw.length; r++) {
        var rowN = napRaw[r];
        var areaNap = rowN[0] ? String(rowN[0]).trim() : "";
        if (areaNap !== "" && areaNap.toUpperCase() !== "TOTAL") {
          napList.push({
            "A": rowN[0],
            "P": rowN[1],
            "H": rowN[2],
            "D1": rowN[3],
            "D3": rowN[4],
            "T": rowN[5]
          });
        }
      }
      resultData = napList;
    }
  }

  // ---------------- SAVE TO CACHE (60 Seconds TTL) ----------------
  var jsonResponse = JSON.stringify(resultData);
  try {
    cache.put("cache_" + type, jsonResponse, 60); // 60s Server Cache
  } catch (err) {
    Logger.log("Caching failed: " + err.message);
  }

  return ContentService.createTextOutput(jsonResponse)
    .setMimeType(ContentService.MimeType.JSON);
}
